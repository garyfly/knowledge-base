## 🌳 技能树（进度条）

> 直接在 [[03 Atlas (MOC)/技能树|技能树]] 里打勾 `- [x]`，下方条会自动重算。

```dataviewjs
const path = "03 Atlas (MOC)/技能树.md";
const file = app.vault.getAbstractFileByPath(path);
if (!file) { dv.paragraph("❌ 找不到技能树笔记，请确认路径：" + path); return; }
const content = await app.vault.read(file);
const lines = content.split("\n");

let current = null;
const skills = {};
for (const line of lines) {
  const h = line.match(/^##\s+(.+)$/);
  if (h) { current = h[1].trim(); skills[current] = { done: 0, total: 0 }; continue; }
  const cb = line.match(/^\s*-\s*\[([ xX])\]/);
  if (cb && current) {
    skills[current].total++;
    if (cb[1].toLowerCase() === "x") skills[current].done++;
  }
}

let totalDone = 0, totalAll = 0;
for (const k in skills) { totalDone += skills[k].done; totalAll += skills[k].total; }

function bar(name, done, total) {
  const pct = total ? Math.round(done / total * 100) : 0;
  const tier = pct < 34 ? "low" : (pct < 67 ? "mid" : "");
  return `<div class="skill-row"><div class="skill-name"><span>${name}</span><span class="pct">${pct}% (${done}/${total})</span></div>` +
         `<div class="game-progress ${tier}"><div class="game-progress-fill" style="width:${pct}%"></div></div></div>`;
}

let html = "";
for (const k in skills) { html += bar(k, skills[k].done, skills[k].total); }
const totalPct = totalAll ? Math.round(totalDone / totalAll * 100) : 0;
html += `<div class="skill-row"><div class="skill-name"><span>🌟 总进度</span><span class="pct">${totalPct}% (${totalDone}/${totalAll})</span></div>` +
        `<div class="game-progress"><div class="game-progress-fill" style="width:${totalPct}%"></div></div></div>`;

dv.container.innerHTML = html;
```
